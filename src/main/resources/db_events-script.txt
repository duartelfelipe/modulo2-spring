-- Criação do Banco de Dados (opcional)
CREATE DATABASE IF NOT EXISTS db_events;
USE db_events;

-- 1. Tabela de Conferências
CREATE TABLE tbl_conference (
    id_conference INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    address VARCHAR(255)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2. Tabela de Sessões (vinculada à conferência)
CREATE TABLE tbl_session (
    session_id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    start_date DATE NOT NULL,
    start_time TIME NOT NULL,
    tbl_conference_id_conference INT NOT NULL,
    CONSTRAINT fk_session_conference 
        FOREIGN KEY (tbl_conference_id_conference) 
        REFERENCES tbl_conference(id_conference) 
        ON DELETE CASCADE 
        ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 3. Tabela de Usuários
CREATE TABLE tbl_user (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    user_name VARCHAR(255) NOT NULL,
    user_email VARCHAR(255) NOT NULL UNIQUE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 4. Tabela de Inscrições (Relação N:N entre Usuários e Sessões/Eventos)
CREATE TABLE tbl_subscription (
    subscribed_user_id INT NOT NULL,
    event_id INT NOT NULL,
    PRIMARY KEY (subscribed_user_id, event_id),
    CONSTRAINT fk_subscription_user 
        FOREIGN KEY (subscribed_user_id) 
        REFERENCES tbl_user(user_id) 
        ON DELETE CASCADE 
        ON UPDATE CASCADE,
    CONSTRAINT fk_subscription_session 
        FOREIGN KEY (event_id) 
        REFERENCES tbl_session(session_id) 
        ON DELETE CASCADE 
        ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
